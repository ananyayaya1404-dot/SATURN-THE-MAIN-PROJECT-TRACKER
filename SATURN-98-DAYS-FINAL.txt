  const days=Math.floor(ms/86400000);ms%=86400000;
  const hours=Math.floor(ms/3600000);ms%=3600000;
  const mins=Math.floor(ms/60000);ms%=60000;
  const secs=Math.floor(ms/1000);
  return {text:`${pad(days)}:${pad(hours)}:${pad(mins)}:${pad(secs)}`,label};
}

function connectionAgeText(){
  if(!connectionLastSeen)return "A MIN AGO";
  const mins=Math.max(0,Math.floor((Date.now()-connectionLastSeen)/60000));
  if(mins<1)return "JUST NOW";
  if(mins===1)return "1 MIN AGO";
  if(mins<60)return `${mins} MINS AGO`;
  const hrs=Math.floor(mins/60);
  if(hrs===1)return "1 HOUR AGO";
  if(hrs<24)return `${hrs} HOURS AGO`;
  const days=Math.floor(hrs/24);
  return days===1?"1 DAY AGO":`${days} DAYS AGO`;
}

function lockPreviousDays(){
  const raw=rawCurrentDay();
  if(raw<=1)return false;
  let changed=false;
  for(let i=1;i<raw&&i<=CONFIG.TOTAL_DAYS;i++){
    const d=data.days[i];if(!d)continue;
    if(!d.locked){
      d.locked=true;
      d.permanentlySaved=true;
      d.completedAt=d.completedAt||new Date().toISOString();
      d.permanentJournal=d.tempJournal||d.permanentJournal||"";
      d.activity=Array.isArray(d.activity)?d.activity:[];
      d.activity.unshift({label:"Day automatically locked at midnight",at:new Date().toISOString()});
      changed=true;
    }
  }
  if(changed){saveLocal();pushOnlineState()}
  return changed;
}

async function connectRealtime(){
  if(!CONFIG.SUPABASE_URL || !CONFIG.SUPABASE_KEY || !window.supabase){onlineMode=false;return}
  try{
    supabaseClient=window.supabase.createClient(CONFIG.SUPABASE_URL,CONFIG.SUPABASE_KEY);
    onlineMode=true;

    // Load the shared tracker state first, so a new phone/browser sees
    // the same SATURN data instead of starting with its own local copy.
    const {data:row,error:loadError}=await supabaseClient
      .from("saturn_state").select("state").eq("id","main").maybeSingle();
    if(!loadError && row?.state){
      data=normalizeData(row.state);
      saveLocal();
    }else if(!loadError && role==="owner"){
      await persistSharedState();
    }

    realtimeChannel=supabaseClient.channel("saturn-98-days-room",{
      config:{presence:{key:role}}
    });
    realtimeChannel
      .on("broadcast",{event:"state"},({payload})=>{
        if(payload?.data){
          data=normalizeData(payload.data);saveLocal();renderAll();
        }
      })
      .on("broadcast",{event:"love"},({payload})=>{if(payload)receiveReaction(payload)})
      .on("broadcast",{event:"presence_last_seen"},({payload})=>{
        if(role==="owner" && payload?.at){
          viewerOnline=false;
          viewerLastSeen=new Date(payload.at).getTime();
          data.viewerLastSeen=new Date(viewerLastSeen).toISOString();
          saveLocal();
          persistSharedState();
          renderViewerPresence();
        }
      })
      .on("presence",{event:"sync"},updatePresence)
      .on("presence",{event:"join"},updatePresence)
      .on("presence",{event:"leave"},updatePresence)
      .subscribe(async status=>{
        if(status==="SUBSCRIBED"){
          renderConnection();
          const nowIso=new Date().toISOString();
          await realtimeChannel.track({role,online_at:nowIso});
          if(role==="viewer"){
            viewerLastSeen=Date.now();
            data.viewerLastSeen=nowIso;
            saveLocal();
          }
          updatePresence();
          await persistSharedState();
        }
      });
  }catch(e){
    onlineMode=false;renderConnection();
    console.warn("Supabase unavailable; using local mode.",e);
  }
}
function elapsedPresenceLabel(ts){
  if(!ts)return "OFFLINE";
  const mins=Math.max(0,Math.floor((Date.now()-ts)/60000));
  if(mins<1)return "JUST NOW";
  if(mins===1)return "1 MIN AGO";
  if(mins<60)return `${mins} MINS AGO`;
  const hrs=Math.floor(mins/60);
  if(hrs===1)return "1 HOUR AGO";
  if(hrs<24)return `${hrs} HOURS AGO`;
  const days=Math.floor(hrs/24);
  return days===1?"1 DAY AGO":`${days} DAYS AGO`;
}

function updatePresence(){
  if(!realtimeChannel)return;
  const state=realtimeChannel.presenceState();
  const viewers=Object.values(state).flat().filter(x=>x.role==="viewer");
  viewerOnline=viewers.length>0;
  if(viewerOnline){
    const stamp=viewers[0]?.online_at;
    if(stamp){
      viewerLastSeen=new Date(stamp).getTime();
      data.viewerLastSeen=new Date(viewerLastSeen).toISOString();
    }
  }else if(!viewerLastSeen && data.viewerLastSeen){
    viewerLastSeen=new Date(data.viewerLastSeen).getTime();
  }
  renderViewerPresence();
}

async function persistSharedState(){
  if(!onlineMode||!supabaseClient)return;
  try{
    const {error}=await supabaseClient.from("saturn_state").upsert({
      id:"main",state:data,updated_at:new Date().toISOString()
    });
    if(error) throw error;
  }catch(e){console.warn("Shared state save failed:",e)}
}
function pushOnlineState(){
  if(!onlineMode||!realtimeChannel)return;
  realtimeChannel.send({type:"broadcast",event:"state",payload:{data}});
  persistSharedState();
}
async function leaveRealtime(){
  if(realtimeChannel){try{await realtimeChannel.untrack();await supabaseClient.removeChannel(realtimeChannel)}catch(e){}}
  realtimeChannel=null;
}
